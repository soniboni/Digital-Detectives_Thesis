#!/usr/bin/env python3
"""
Phase 6: Model Application and Output Generation
Standalone script for NTFS timestomping detection

This script applies the trained LightGBM model to file features
and generates detection reports.

Usage:
    python apply_model.py --input <file_features.csv> --output <output_dir>

Outputs:
    - detected_files.csv: Files flagged as timestomped with explanations
    - files_with_features.csv: All files with confidence scores
    - summary.txt: Human-readable detection report
"""

import argparse
import pandas as pd
import numpy as np
from pathlib import Path
from datetime import datetime
import joblib
import json
import sys


# =============================================================================
# CONFIGURATION
# =============================================================================

# Default paths (can be overridden via command line)
DEFAULT_MODEL_PATH = Path(__file__).parent.parent / "data/Phase 4: Model Training/final version/model_lightgbm.joblib"
DEFAULT_CONFIG_PATH = Path(__file__).parent.parent / "data/Phase 4: Model Training/final version/training_config.json"

# Detection threshold (from Phase 4)
DETECTION_THRESHOLD = 0.02

# Confidence levels
CONFIDENCE_HIGH = 0.5
CONFIDENCE_MEDIUM = 0.1


# =============================================================================
# DETECTION EXPLANATION FUNCTIONS
# =============================================================================

def explain_detection(row, feature_columns):
    """
    Generate human-readable explanation for why a file was flagged.
    Returns a list of detection reasons based on feature values.
    """
    explanations = []
    
    # Backward timestamp jumps (strong indicator)
    if row.get('num_backward_jumps', 0) > 0:
        jump_sec = row.get('max_backward_jump_seconds', 0)
        explanations.append(
            f"BACKWARD_TIMESTAMP: {int(row['num_backward_jumps'])} backward jump(s), "
            f"max {jump_sec:.0f} seconds"
        )
    
    # Zero nanoseconds (timestomping tools often set round timestamps)
    if row.get('num_zero_nanosecond_events', 0) > 0:
        explanations.append(
            f"ZERO_NANOSECONDS: {int(row['num_zero_nanosecond_events'])} event(s) "
            f"with zero nanosecond precision"
        )
    
    # SI-only modifications
    if row.get('only_SI_modified', 0) == 1:
        explanations.append(
            "SI_ONLY_MODIFICATION: Only $STANDARD_INFORMATION modified, $FILE_NAME unchanged"
        )
    
    # LogFile/USN mismatch
    if row.get('logfile_usn_mismatch', 0) > 0:
        explanations.append(
            "ARTIFACT_MISMATCH: LogFile and UsnJrnl timestamps do not correlate"
        )
    
    # Repeated update operations
    if row.get('repeated_update_resident_value', 0) == 1:
        explanations.append(
            "REPEATED_UPDATES: Multiple rapid UpdateResidentValue operations"
        )
    
    # Timestamp changes
    if row.get('num_timestamp_changes', 0) > 2:
        explanations.append(
            f"TIMESTAMP_CHANGES: {int(row['num_timestamp_changes'])} modification event(s)"
        )
    
    # Creation time changes
    if row.get('num_creation_changes', 0) > 0:
        explanations.append(
            f"CREATION_MODIFIED: {int(row['num_creation_changes'])} creation time change(s)"
        )
    
    # USN basic pattern
    if row.get('has_usn_basic_pattern', 0) == 1:
        explanations.append(
            "USN_PATTERN: Basic_Info + Close without File_Create"
        )
    
    if not explanations:
        explanations.append("LOW_CONFIDENCE: Flagged by combination of minor indicators")
    
    return "; ".join(explanations)


def get_confidence_level(probability):
    """Categorize confidence level."""
    if probability >= CONFIDENCE_HIGH:
        return "HIGH"
    elif probability >= CONFIDENCE_MEDIUM:
        return "MEDIUM"
    else:
        return "LOW"


# =============================================================================
# MAIN DETECTION FUNCTION
# =============================================================================

def apply_model(input_path, output_dir, model_path=None, config_path=None, threshold=None):
    """
    Apply trained model to file features and generate output reports.
    
    Parameters:
        input_path: Path to file_features.csv from Phase 3
        output_dir: Directory to save output files
        model_path: Path to trained model (optional, uses default)
        config_path: Path to training config (optional, uses default)
        threshold: Detection threshold (optional, uses default 0.02)
    
    Returns:
        Dictionary with detection statistics
    """
    # Set defaults
    if model_path is None:
        model_path = DEFAULT_MODEL_PATH
    if config_path is None:
        config_path = DEFAULT_CONFIG_PATH
    if threshold is None:
        threshold = DETECTION_THRESHOLD
    
    # Convert to Path objects
    input_path = Path(input_path)
    output_dir = Path(output_dir)
    model_path = Path(model_path)
    config_path = Path(config_path)
    
    # Create output directory
    output_dir.mkdir(parents=True, exist_ok=True)
    
    print("=" * 70)
    print("NTFS TIMESTOMPING DETECTION - MODEL APPLICATION")
    print("=" * 70)
    
    # -------------------------------------------------------------------------
    # Load Model and Configuration
    # -------------------------------------------------------------------------
    print(f"\nLoading model from: {model_path}")
    if not model_path.exists():
        raise FileNotFoundError(f"Model not found: {model_path}")
    model = joblib.load(model_path)
    
    print(f"Loading config from: {config_path}")
    if not config_path.exists():
        raise FileNotFoundError(f"Config not found: {config_path}")
    with open(config_path, 'r') as f:
        config = json.load(f)
    
    feature_columns = config['feature_columns']
    print(f"Feature columns: {len(feature_columns)}")
    
    # -------------------------------------------------------------------------
    # Load Input Data
    # -------------------------------------------------------------------------
    print(f"\nLoading features from: {input_path}")
    if not input_path.exists():
        raise FileNotFoundError(f"Input file not found: {input_path}")
    
    df = pd.read_csv(input_path, low_memory=False)
    print(f"Total files to analyze: {len(df):,}")
    
    # -------------------------------------------------------------------------
    # Prepare Features
    # -------------------------------------------------------------------------
    print("\nPreparing features...")
    
    # Check for missing columns
    missing_cols = [col for col in feature_columns if col not in df.columns]
    if missing_cols:
        print(f"WARNING: Missing columns: {missing_cols}")
        for col in missing_cols:
            df[col] = 0
    
    X = df[feature_columns].copy()
    X = X.fillna(0)
    
    # Convert object columns to numeric
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col].map({
                'True': 1, 'False': 0, True: 1, False: 0
            }), errors='coerce').fillna(0)
    
    # Convert boolean to int
    bool_cols = X.select_dtypes(include=['bool']).columns
    X[bool_cols] = X[bool_cols].astype(int)
    
    # Ensure float type
    X = X.astype(float)
    
    # -------------------------------------------------------------------------
    # Apply Model
    # -------------------------------------------------------------------------
    print("\nApplying model...")
    start_time = datetime.now()
    
    y_prob = model.predict_proba(X.values)[:, 1]
    y_pred = (y_prob >= threshold).astype(int)
    
    elapsed = (datetime.now() - start_time).total_seconds()
    print(f"Prediction complete in {elapsed:.2f} seconds")
    
    # Add predictions to dataframe
    df['Confidence'] = y_prob
    df['Flagged'] = y_pred
    df['Confidence_Level'] = df['Confidence'].apply(get_confidence_level)
    
    # -------------------------------------------------------------------------
    # Generate Detection Explanations
    # -------------------------------------------------------------------------
    print("\nGenerating detection explanations...")
    
    detected_mask = df['Flagged'] == 1
    df.loc[detected_mask, 'Detection_Reasons'] = df[detected_mask].apply(
        lambda row: explain_detection(row, feature_columns), axis=1
    )
    df['Detection_Reasons'] = df['Detection_Reasons'].fillna('')
    
    # -------------------------------------------------------------------------
    # Calculate Statistics
    # -------------------------------------------------------------------------
    total_files = len(df)
    flagged_files = detected_mask.sum()
    high_conf = (df['Confidence'] >= CONFIDENCE_HIGH).sum()
    medium_conf = ((df['Confidence'] >= CONFIDENCE_MEDIUM) & (df['Confidence'] < CONFIDENCE_HIGH)).sum()
    low_conf = ((df['Confidence'] >= threshold) & (df['Confidence'] < CONFIDENCE_MEDIUM)).sum()
    
    crr = 1 - (flagged_files / total_files) if total_files > 0 else 0
    flag_rate = flagged_files / total_files if total_files > 0 else 0
    
    stats = {
        'total_files': total_files,
        'flagged_files': flagged_files,
        'high_confidence': high_conf,
        'medium_confidence': medium_conf,
        'low_confidence': low_conf,
        'crr': crr,
        'flag_rate': flag_rate,
        'threshold': threshold
    }
    
    print(f"\nDetection Statistics:")
    print(f"  Total files analyzed: {total_files:,}")
    print(f"  Files flagged: {flagged_files:,} ({flag_rate*100:.2f}%)")
    print(f"  High confidence (>0.5): {high_conf:,}")
    print(f"  Medium confidence (0.1-0.5): {medium_conf:,}")
    print(f"  Low confidence (0.02-0.1): {low_conf:,}")
    print(f"  Candidate Reduction Rate: {crr*100:.2f}%")
    
    # -------------------------------------------------------------------------
    # Output 1: detected_files.csv
    # -------------------------------------------------------------------------
    print("\nGenerating detected_files.csv...")
    
    detected_df = df[detected_mask].copy()
    detected_df = detected_df.sort_values('Confidence', ascending=False)
    
    # Select columns for output
    output_cols = ['FileName', 'FilePath', 'Confidence', 'Confidence_Level', 'Detection_Reasons']
    
    # Add key feature columns
    key_features = [
        'num_backward_jumps', 'max_backward_jump_seconds', 
        'num_zero_nanosecond_events', 'only_SI_modified',
        'num_timestamp_changes', 'logfile_usn_mismatch'
    ]
    for col in key_features:
        if col in detected_df.columns:
            output_cols.append(col)
    
    detected_output = detected_df[[c for c in output_cols if c in detected_df.columns]]
    detected_path = output_dir / "detected_files.csv"
    detected_output.to_csv(detected_path, index=False)
    print(f"  Saved: {detected_path} ({len(detected_output):,} files)")
    
    # -------------------------------------------------------------------------
    # Output 2: files_with_features.csv
    # -------------------------------------------------------------------------
    print("\nGenerating files_with_features.csv...")
    
    # All files with confidence scores
    all_files_df = df.copy()
    all_files_df = all_files_df.sort_values('Confidence', ascending=False)
    
    files_path = output_dir / "files_with_features.csv"
    all_files_df.to_csv(files_path, index=False)
    print(f"  Saved: {files_path} ({len(all_files_df):,} files)")
    
    # -------------------------------------------------------------------------
    # Output 3: summary.txt
    # -------------------------------------------------------------------------
    print("\nGenerating summary.txt...")
    
    summary_lines = []
    summary_lines.append("=" * 80)
    summary_lines.append("NTFS TIMESTOMPING DETECTION REPORT")
    summary_lines.append("=" * 80)
    summary_lines.append("")
    summary_lines.append(f"Analysis Date:      {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    summary_lines.append(f"Input File:         {input_path.name}")
    summary_lines.append(f"Model Used:         LightGBM (Phase 4 Baseline)")
    summary_lines.append(f"Threshold:          {threshold}")
    summary_lines.append("")
    summary_lines.append("-" * 80)
    summary_lines.append("SUMMARY")
    summary_lines.append("-" * 80)
    summary_lines.append(f"Total Files Analyzed:       {total_files:,}")
    summary_lines.append(f"Files Flagged:              {flagged_files:,} ({flag_rate*100:.2f}%)")
    summary_lines.append(f"High Confidence (>0.5):     {high_conf:,}")
    summary_lines.append(f"Medium Confidence (0.1-0.5): {medium_conf:,}")
    summary_lines.append(f"Low Confidence (0.02-0.1):  {low_conf:,}")
    summary_lines.append("")
    summary_lines.append(f"Candidate Reduction Rate:   {crr*100:.2f}%")
    summary_lines.append(f"  (Analyst reviews {flagged_files:,} instead of {total_files:,} files)")
    summary_lines.append("")
    
    # Top 20 most suspicious files
    summary_lines.append("-" * 80)
    summary_lines.append("TOP 20 MOST SUSPICIOUS FILES")
    summary_lines.append("-" * 80)
    summary_lines.append(f"{'Rank':<6} {'Confidence':<12} {'FilePath'}")
    summary_lines.append("-" * 80)
    
    top_files = detected_df.head(20)
    for rank, (_, row) in enumerate(top_files.iterrows(), 1):
        filepath = row.get('FilePath', row.get('FileName', 'Unknown'))
        if len(filepath) > 55:
            filepath = "..." + filepath[-52:]
        summary_lines.append(f"{rank:<6} {row['Confidence']:<12.4f} {filepath}")
    
    summary_lines.append("")
    
    # Detection pattern summary
    summary_lines.append("-" * 80)
    summary_lines.append("DETECTION PATTERN SUMMARY (among flagged files)")
    summary_lines.append("-" * 80)
    
    if len(detected_df) > 0:
        patterns = {
            'Backward timestamp jumps': (detected_df['num_backward_jumps'] > 0).sum() if 'num_backward_jumps' in detected_df.columns else 0,
            'Zero nanosecond timestamps': (detected_df['num_zero_nanosecond_events'] > 0).sum() if 'num_zero_nanosecond_events' in detected_df.columns else 0,
            'SI-only modifications': (detected_df['only_SI_modified'] == 1).sum() if 'only_SI_modified' in detected_df.columns else 0,
            'LogFile/USN mismatch': (detected_df['logfile_usn_mismatch'] > 0).sum() if 'logfile_usn_mismatch' in detected_df.columns else 0,
            'Multiple timestamp changes': (detected_df['num_timestamp_changes'] > 2).sum() if 'num_timestamp_changes' in detected_df.columns else 0,
        }
        
        for pattern, count in sorted(patterns.items(), key=lambda x: x[1], reverse=True):
            pct = count / len(detected_df) * 100
            summary_lines.append(f"  {pattern:<35} {count:>8,} ({pct:>5.1f}%)")
    
    summary_lines.append("")
    
    # Methodology
    summary_lines.append("-" * 80)
    summary_lines.append("METHODOLOGY")
    summary_lines.append("-" * 80)
    summary_lines.append("This analysis uses machine learning (LightGBM) trained on 23 datasets with")
    summary_lines.append("52 known timestomped files. The model achieves 100% recall on validation")
    summary_lines.append("datasets, ensuring no known timestomped files are missed.")
    summary_lines.append("")
    summary_lines.append(f"Features analyzed: {len(feature_columns)}")
    summary_lines.append("Based on: Oh et al. (2024) NTFS timestomping detection methodology")
    summary_lines.append("")
    summary_lines.append("=" * 80)
    summary_lines.append("Generated by Digital Detectives Timestomping Detector v1.0")
    summary_lines.append("=" * 80)
    
    summary_path = output_dir / "summary.txt"
    with open(summary_path, 'w') as f:
        f.write('\n'.join(summary_lines))
    print(f"  Saved: {summary_path}")
    
    # -------------------------------------------------------------------------
    # Done
    # -------------------------------------------------------------------------
    print("\n" + "=" * 70)
    print("MODEL APPLICATION COMPLETE")
    print("=" * 70)
    print(f"\nOutput files saved to: {output_dir}")
    print(f"  - detected_files.csv ({flagged_files:,} files)")
    print(f"  - files_with_features.csv ({total_files:,} files)")
    print(f"  - summary.txt")
    
    return stats


# =============================================================================
# COMMAND LINE INTERFACE
# =============================================================================

def main():
    parser = argparse.ArgumentParser(
        description="Apply trained LightGBM model to detect NTFS timestomping"
    )
    parser.add_argument(
        '--input', '-i',
        required=True,
        help='Path to file_features.csv from Phase 3'
    )
    parser.add_argument(
        '--output', '-o',
        required=True,
        help='Output directory for results'
    )
    parser.add_argument(
        '--model', '-m',
        default=None,
        help='Path to trained model (default: Phase 4 model)'
    )
    parser.add_argument(
        '--config', '-c',
        default=None,
        help='Path to training config (default: Phase 4 config)'
    )
    parser.add_argument(
        '--threshold', '-t',
        type=float,
        default=0.02,
        help='Detection threshold (default: 0.02)'
    )
    
    args = parser.parse_args()
    
    try:
        stats = apply_model(
            input_path=args.input,
            output_dir=args.output,
            model_path=args.model,
            config_path=args.config,
            threshold=args.threshold
        )
        return 0
    except Exception as e:
        print(f"\nERROR: {e}", file=sys.stderr)
        return 1


if __name__ == "__main__":
    sys.exit(main())
