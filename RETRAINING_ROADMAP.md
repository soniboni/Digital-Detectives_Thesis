## Critical Discovery: Suspicious CSVs Structure

### Raw CSV Column Structure

**LogFile CSVs**:
```
Columns: LSN, EventTime(UTC+8), Event, Detail, File/Directory Name, Full Path,
         CreationTime, ModifiedTime, MFTModifiedTime, AccessedTime, ...
```
- Has `Detail` field with forensic information

**UsnJrnl CSVs**:
```
Columns: TimeStamp(UTC+8), USN, File/Directory Name, FullPath, EventInfo,
         SourceInfo, FileAttribute, ...
```
- **NO Detail field** in raw UsnJrnl CSVs!
- Only has `EventInfo` (e.g., "File_Closed", "Basic_Info_Changed")

**Suspicious CSVs (Ground Truth)**:
```
Columns: source, lsn/usn, category, detail
```
- `source`: "logfile" or "usnjrnl"
- `lsn/usn`: LSN (if logfile) or USN (if usnjrnl) - used for mapping
- `category`: "Timestamp Manipulation", "Execution of Suspicious Programs", etc.
- `detail`: **Contains forensic patterns including "Zero in 100-nanoseconds"**

### Key Insight

The "Zero in 100-nanoseconds" pattern for UsnJrnl-only files is ONLY available in Suspicious CSVs (generated by NTFS Artifact Analysis Tool). This means:

1. We must incorporate Suspicious CSVs into our merge
2. Map suspicious records to raw records by LSN/USN
3. Extract zero_in_nanoseconds from `suspicious_detail` field

### 1.3: Ground Truth Labeling

```python
def create_ground_truth_label(row):
    """
    Label: 1 = timestomped, 0 = benign/tunneling

    Timestomped if:
    - Flagged in Suspicious CSV
    - Has suspicious_category = "Timestamp Manipulation"
    """
    if row['is_flagged_suspicious']:
        # Check if it's actual timestomping (not other categories)
        categories = [
            str(row.get('suspicious_category_lf', '')),
            str(row.get('suspicious_category_usn', ''))
        ]

        for cat in categories:
            if 'Timestamp Manipulation' in cat:
                return 1  # TRUE timestomping

    return 0  # Benign

df_merged['predicted_timestomped'] = df_merged.apply(
    create_ground_truth_label, axis=1
)
```

## Phase 2: Feature Engineering

### 2.1: Fix Zero Nanoseconds Extraction (CRITICAL)

```python
def extract_zero_nanoseconds(df):
    """
    Extract zero nanoseconds pattern from suspicious_detail fields.

    CRITICAL: Raw UsnJrnl CSVs don't have detail field!
    Must check suspicious_detail from Suspicious CSV merge.
    """
    # Check LogFile suspicious detail
    lf_zero_nano = df['suspicious_detail_lf'].fillna('').str.contains(
        'Zero in 100-nanoseconds', case=False, na=False
    )

    # Check UsnJrnl suspicious detail (CRITICAL FIX!)
    usn_zero_nano = df['suspicious_detail_usn'].fillna('').str.contains(
        'Zero in 100-nanoseconds', case=False, na=False
    )

    # Combine: True if EITHER source shows zero nanoseconds
    df['zero_in_nanoseconds'] = lf_zero_nano | usn_zero_nano

    print(f"Zero nanoseconds detected:")
    print(f"  From LogFile: {lf_zero_nano.sum():,} events")
    print(f"  From UsnJrnl: {usn_zero_nano.sum():,} events")
    print(f"  Total: {df['zero_in_nanoseconds'].sum():,} events")

    return df
```

### 2.2: Cross-Artifact Validation Features

```python
def extract_cross_artifact_features(df):
    """
    Calculate cross-artifact validation score.
    """
    def calc_score(row):
        if row['source'] == 'both':
            # Both artifacts present
            lf_has_time_reversal = 'Time Reversal' in str(row.get('Event_lf', ''))
            usn_has_basic_info = 'Basic_Info_Change' in str(row.get('EventInfo_usn', ''))

            if lf_has_time_reversal and usn_has_basic_info:
                return 3  # Highest confidence
            else:
                return 2
        elif row['source'] == 'logfile_only':
            return 2 if 'Time Reversal' in str(row.get('Event_lf', '')) else 1
        elif row['source'] == 'usnjrnl_only':
            return 1 if 'Basic_Info_Change' in str(row.get('EventInfo_usn', '')) else 0
        return 0

    df['cross_artifact_validation_score'] = df.apply(calc_score, axis=1)

    # Evidence flags
    df['has_logfile_evidence'] = df['Event_lf'].notna()
    df['has_usnjrnl_evidence'] = df['EventInfo_usn'].notna()

    return df
```

### 2.3: Temporal & Behavioral Features

```python
def extract_temporal_features(df):
    """Extract event clustering and frequency features."""
    df = df.sort_values(['case_id', 'EventTime']).reset_index(drop=True)

    # Event frequency per file
    df['event_frequency_per_file'] = df.groupby(
        ['case_id', 'filename', 'filepath']
    )['filename'].transform('count')

    # Events in temporal windows
    df['events_in_1min_window'] = 0
    df['events_in_5min_window'] = 0

    for case_id in df['case_id'].unique():
        case_df = df[df['case_id'] == case_id]
        event_times = pd.to_datetime(case_df['EventTime']).values

        for idx in case_df.index:
            current_time = pd.to_datetime(df.at[idx, 'EventTime'])

            # 1-minute window (±30s)
            window_1min = (
                (event_times >= current_time - pd.Timedelta(seconds=30)) &
                (event_times <= current_time + pd.Timedelta(seconds=30))
            )
            df.at[idx, 'events_in_1min_window'] = window_1min.sum() - 1

            # 5-minute window (±2.5min)
            window_5min = (
                (event_times >= current_time - pd.Timedelta(minutes=2.5)) &
                (event_times <= current_time + pd.Timedelta(minutes=2.5))
            )
            df.at[idx, 'events_in_5min_window'] = window_5min.sum() - 1

    return df
```

### 2.4: File Characteristics Features

```python
def extract_file_characteristics(df):
    """Extract file type and path features."""
    # Executable detection
    exec_exts = ['.exe', '.dll', '.sys', '.com', '.scr', '.bat', '.ps1']
    df['is_executable'] = df['filename'].fillna('').str.lower().apply(
        lambda x: any(x.endswith(ext) for ext in exec_exts)
    )

    # Archive detection
    archive_exts = ['.zip', '.rar', '.7z', '.tar', '.gz']
    df['is_archive'] = df['filename'].fillna('').str.lower().apply(
        lambda x: any(x.endswith(ext) for ext in archive_exts)
    )

    # Path depth
    df['path_depth'] = df['filepath'].fillna('').str.count(r'[\\|/]')

    # Filename length
    df['filename_length'] = df['filename'].fillna('').str.len()

    # Windows directory check
    df['in_windows_dir'] = df['filepath'].fillna('').str.contains(
        r'\\Windows\\', case=False, na=False
    )

    return df
```

```python
class TimestompDetectorIngestModule(DataSourceIngestModule):
    def process(self, dataSource, progressBar):
        # Extract $LogFile records
        logfile_records = self.extract_logfile(dataSource)

        # Extract $UsnJrnl records
        usnjrnl_records = self.extract_usnjrnl(dataSource)

        # Merge artifacts
        merged = self.smart_merge(logfile_records, usnjrnl_records)

        # Extract features
        features = self.extract_features(merged)

        # Run detection
        predictions = self.model.predict_proba(features)[:, 1]

        # Flag high-confidence detections
        for idx, conf in enumerate(predictions):
            if conf >= 0.70:
                self.flag_timestomped_file(merged.iloc[idx], confidence=conf)

        return IngestModule.ProcessResult.OK
```
