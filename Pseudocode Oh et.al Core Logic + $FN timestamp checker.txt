- **Algorithm 2: Identification of the Event Target File**
    
    ```jsx
    fucntion GetTargetFile(record) 
    	if There is a $MFT entry having LSN same as record.LSN then 
    		file.file_name <- $MFTEntry.file_name
    		file.file_path <- $MFTEntry.file_path
    		return file 
    	end if 
    	i <- 1 
    	while file_level_event_count-i >= 0 do 
    		previous_event <- arr_file_level_events[file_level_event_count-i]
    		if previous_event.target_vcn == record.target_vcn 
    		&& previous_event.mft_cluster_index == record.mft_cluster_index then 
    			if previous.event.event_type == "File Creation" 
    			|| previous.event.event_type == "Renaming File" 
    			|| previous.event.event_type = "File Move" then 
    				file.name <- previous.event.file_name 
    				file.path <- previous.event.file_path 
    				return file 
    			else if previous_event.event_type == "File Deletion" then 
    				break 
    			end if 
    		end if 
    	end while 
    	entry_number <- record.target_vcn * 4 
    	entry_number <- entry_number + (record.mft_cluster_index / 2) 
    	if There is a $MFT entry located at entry_number then 
    		if the $MFT entry is not inactive then 
    			file.name <- $MFTEntry.file_name 
    			file.path <- $MFTEntry.file_path 
    			return file 
    		end if 
    	end if 
    	return NULL 
    end funtion
    ```

    ```jsx
    	last_basic_detection_pattern_time <- record.timestamp

    		end if
    ```
